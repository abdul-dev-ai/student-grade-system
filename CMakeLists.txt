cmake_minimum_required(VERSION 3.16)
project(StudentGradeSystem VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(grade_core src/grade_engine.cpp)
target_include_directories(grade_core PUBLIC include)
if(MSVC)
  target_compile_options(grade_core PRIVATE /W4)
else()
  target_compile_options(grade_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(EMSCRIPTEN)
  add_executable(grade_engine src/bindings.cpp)
  target_link_libraries(grade_engine PRIVATE grade_core)
  set_target_properties(grade_engine PROPERTIES SUFFIX ".mjs")
  target_link_options(grade_engine PRIVATE --bind --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web,node -sFILESYSTEM=0)
else()
  enable_testing()
  add_executable(grade_tests tests/grade_engine_test.cpp)
  target_link_libraries(grade_tests PRIVATE grade_core)
  add_test(NAME grade_engine_tests COMMAND grade_tests)
endif()
